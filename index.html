document.getElementById('uploadButton').addEventListener('click', async () => { const fileInput = document.getElementById('uploadInput'); if (fileInput.files.length === 0) { alert('請先選擇一張照片！'); return; } const file = fileInput.files[0]; const reader = new FileReader(); reader.readAsDataURL(file); reader.onload = async function () { const base64Data = reader.result; const payload = { file: base64Data, filename: file.name, mimeType: file.type }; // 將這行替換為你剛剛提供的 Google Apps Script 網址 const scriptURL = 'https://script.google.com/macros/s/AKfycbwWcGqlhfs4tIHizuhvk2M-fBR5JCu_Rj0UNBp-Na3zkueEdGF07KxrCKjzzvZ8yiEwFg/exec'; try { alert('正在上傳至您的 Google 雲端，請稍候...'); const response = await fetch(scriptURL, { method: 'POST', body: JSON.stringify(payload) }); const result = await response.json(); if (result.success) { const imageUrl = result.url; console.log('上傳成功！圖片網址：', imageUrl); // 【提示】拿到 imageUrl 後，你可以把它存入 localStorage 或你的網頁資料結構中 // 例如：document.getElementById('myImage').src = imageUrl; alert('照片已成功上傳並存入你的 Google 雲端資料夾！'); } else { alert('上傳失敗：' + result.error); } } catch (error) { console.error('發生錯誤：', error); alert('上傳發生網路錯誤'); } }; });
